Panduan Setup Backend Google Apps Script (GAS)

Langkah Integrasi Realtime Google Sheets:

  1. Buat Google Spreadsheet baru di Google Drive Anda.
  2. Buka menu Ekstensi > Apps Script.
  3. Salin seluruh kode script di bawah ini ke dalam editor Apps Script.
  4. Klik Simpan lalu klik Deploy > Deployment Baru.
  5. Pilih Jenis: Aplikasi Web, Akses: Siapa Saja (Anyone), lalu Deploy.
  6. Tempelkan Web App URL yang dihasilkan ke input box di atas.
// ==========================================
// BACKEND GOOGLE APPS SCRIPT (INFORMATIKA QR)
// ==========================================

function doGet(e) {
  var action = e.parameter.action;
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  if (action == "getSiswa") {
    var sheet = ss.getSheetByName("Siswa") || setupSheets();
    var data = sheet.getDataRange().getValues();
    var headers = data.shift();
    return responseJSON(data);
  }
  
  if (action == "getAbsensi") {
    var sheet = ss.getSheetByName("Absensi") || setupSheets();
    var data = sheet.getDataRange().getValues();
    return responseJSON(data);
  }
  
  return responseJSON({status: "online", app: "Informatika QR Backend"});
}

function doPost(e) {
  try {
    var contents = JSON.parse(e.postData.contents);
    var action = contents.action;
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    
    if (action == "saveAbsensi") {
      var sheet = ss.getSheetByName("Absensi");
      sheet.appendRow([
        new Date().toISOString(),
        contents.nis,
        contents.nama,
        contents.kelas,
        contents.status,
        contents.catatan || ""
      ]);
      return responseJSON({success: true, message: "Absensi disimpan"});
    }
  } catch(err) {
    return responseJSON({error: err.toString()});
  }
}

function responseJSON(data) {
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

function setupSheets() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheetSiswa = ss.getSheetByName("Siswa") || ss.insertSheet("Siswa");
  if (sheetSiswa.getLastRow() === 0) {
    sheetSiswa.appendRow(["NIS", "Nama", "Kelas", "Gender"]);
  }
  var sheetAbsen = ss.getSheetByName("Absensi") || ss.insertSheet("Absensi");
  if (sheetAbsen.getLastRow() === 0) {
    sheetAbsen.appendRow(["Timestamp", "NIS", "Nama", "Kelas", "Status", "Catatan"]);
  }
  return sheetSiswa;
}
